        function selectTopic(id) {
            currentTopic = discussionData.find(t => t.id === id);
            
            // UI 업데이트
            document.getElementById('current-book-label').innerText = currentTopic.book;
            document.getElementById('current-book-summary').innerText = currentTopic.bookSummary;
            document.getElementById('current-topic-title').innerText = currentTopic.title;
            document.getElementById('current-topic-desc').innerText = currentTopic.desc;
            
            // 상태 초기화
            selectedStance = null;
            document.getElementById('user-opinion-main').value = '';
            document.getElementById('user-evidence').value = '';
            resetStanceButtons();
            document.getElementById('writing-area').classList.add('hidden');

            // 화면 전환
            step1.classList.add('hidden');
            step2.classList.remove('hidden');
            window.scrollTo(0, 0);
        }

        function goBackToStep1() {

            step1.classList.remove('hidden');
        }

        function selectStance(stance) {
            selectedStance = stance;
            resetStanceButtons();
            
            const btnPro = document.getElementById('btn-pro');
            const btnCon = document.getElementById('btn-con');
            const radioPro = btnPro.querySelector('.stance-radio');
            const radioCon = btnCon.querySelector('.stance-radio');

            if (stance === 'pro') {
                btnPro.classList.add('border-blue-500', 'bg-blue-50');
                btnPro.classList.remove('border-gray-200');
                radioPro.classList.add('border-blue-500');
                radioPro.innerHTML = '<div class="w-2.5 h-2.5 bg-blue-500 rounded-full"></div>';
            } else {
                btnCon.classList.add('border-red-500', 'bg-red-50');
                btnCon.classList.remove('border-gray-200');
                radioCon.classList.add('border-red-500');
                radioCon.innerHTML = '<div class="w-2.5 h-2.5 bg-red-500 rounded-full"></div>';
            }

            document.getElementById('writing-area').classList.remove('hidden');
            // 조금 부드럽게 스크롤
            setTimeout(() => {
                document.getElementById('writing-area').scrollIntoView({ behavior: 'smooth', block: 'end' });
            }, 100);
        }

        function resetStanceButtons() {
            const btnPro = document.getElementById('btn-pro');
            const btnCon = document.getElementById('btn-con');
            
            btnPro.className = "flex-1 py-4 px-6 rounded-xl border-2 border-gray-200 hover:border-blue-500 hover:bg-blue-50 transition-all text-left group";
            btnCon.className = "flex-1 py-4 px-6 rounded-xl border-2 border-gray-200 hover:border-red-500 hover:bg-red-50 transition-all text-left group";
            
            btnPro.querySelector('.stance-radio').innerHTML = '';
            btnPro.querySelector('.stance-radio').className = "w-5 h-5 rounded-full border-2 border-gray-300 group-hover:border-blue-500 flex items-center justify-center stance-radio";
            
            btnCon.querySelector('.stance-radio').innerHTML = '';
            btnCon.querySelector('.stance-radio').className = "w-5 h-5 rounded-full border-2 border-gray-300 group-hover:border-red-500 flex items-center justify-center stance-radio";
        }

        function submitOpinion() {
            const userMain = document.getElementById('user-opinion-main').value.trim();
            const userEvidence = document.getElementById('user-evidence').value.trim();
            
            if (!selectedStance) {
                showToast("찬성 또는 반대 입장을 먼저 선택해 주세요.", "error");
                return;
            }
            if (userMain.length < 5) {
                showToast("핵심 주장을 조금 더 명확하게 적어주세요!", "error");
                document.getElementById('user-opinion-main').focus();
                return;
            }
            if (userEvidence.length < 10) {
                showToast("근거를 10자 이상 조금 더 자세히 적어주세요!", "error");
                document.getElementById('user-evidence').focus();
                return;
            }

            // 결과 화면 데이터 바인딩
            const stanceLabel = selectedStance === 'pro' 
                ? '<i class="fa-regular fa-thumbs-up text-blue-500"></i> 찬성' 
                : '<i class="fa-regular fa-thumbs-down text-red-500"></i> 반대';
            
            document.getElementById('result-my-stance').innerHTML = stanceLabel;
            document.getElementById('result-my-opinion').innerText = userMain;
            document.getElementById('result-my-evidence').innerText = userEvidence;

            // 가상 친구 데이터 바인딩
            document.getElementById('friend-pro-name').innerText = currentTopic.friends.pro.name;
            document.getElementById('friend-pro-text').innerText = currentTopic.friends.pro.text;
            
            document.getElementById('friend-con-name').innerText = currentTopic.friends.con.name;
            document.getElementById('friend-con-text').innerText = currentTopic.friends.con.text;

            // 화면 전환
            step2.classList.add('hidden');
            step3.classList.remove('hidden');
            window.scrollTo(0, 0);
            
            showToast("성공적으로 등록되었습니다!", "success");
        }

        function copyMyText() {
            const mainText = document.getElementById('result-my-opinion').innerText;
            const evidenceText = document.getElementById('result-my-evidence').innerText;
            const fullText = `[나의 생각 - ${currentTopic.title}]\n입장: ${selectedStance === 'pro' ? '찬성' : '반대'}\n주장: ${mainText}\n근거: ${evidenceText}`;
            
            // iFrame 호환성을 위해 execCommand 사용
            const textArea = document.createElement("textarea");
            textArea.value = fullText;
            textArea.style.position = "fixed";
            textArea.style.left = "-999999px";
            textArea.style.top = "-999999px";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();

            try {
                const successful = document.execCommand('copy');
                if (successful) {
                    showToast("내 글이 클립보드에 복사되었습니다! 📋", "success");
                } else {
                    showToast("복사에 실패했습니다.", "error");
                }
            } catch (err) {
                showToast("이 브라우저에서는 복사 기능을 지원하지 않습니다.", "error");
            }

            document.body.removeChild(textArea);
        }

        function resetApp() {
            step3.classList.add('hidden');
            step1.classList.remove('hidden');
            window.scrollTo(0, 0);
        }

        let toastTimeout;
        function showToast(message, type) {
            const toast = document.getElementById('toast-message');
            const toastText = document.getElementById('toast-text');
            const icon = toast.querySelector('i');
            
            toastText.innerText = message;
            
            if (type === 'error') {
                icon.className = "fa-solid fa-circle-exclamation text-red-400";
            } else {
                icon.className = "fa-solid fa-circle-check text-green-400";
            }

            toast.classList.add('show');
            
            clearTimeout(toastTimeout);
            toastTimeout = setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }
